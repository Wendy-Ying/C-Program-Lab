#include<stdio.h>
#include<stdlib.h>
#include<string.h>
char* insert(char* ss, char* s1, char* s2, int num);
int main()
{
	char s1[] = "IChina";
	char s2[] = "Love";
	char* ss = NULL;
	insert(ss, s1, s2, 1);
	printf("%s", ss);
	return 0;
}
char* insert(char* ss, char* s1, char* s2, int num)
{
	ss = (char*)malloc(sizeof(s1) + sizeof(s2) + sizeof(char));
	int i = 0;
	for (i = 0; i < num; i++)
	{
		*ss = *s1;
		ss++;
		s1++;
	}
	for (i = 0; i < strlen(s2); i++)
	{
		*ss = *s2;
		ss++;
		s2++;
	}
	for (i = 0; i < strlen(s1) - num; i++)
	{
		*ss = *s1;
		ss++;
		s1++;
	}
	free(ss);
	return 0;
}
