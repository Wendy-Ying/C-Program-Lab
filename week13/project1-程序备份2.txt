#define _CRT_SECURE_NO_WARNINGS
#include <iostream>
#include <opencv2/opencv.hpp>
#include<stdio.h>
#include<stdlib.h>
#include<math.h>
#include<string.h>
using namespace std;
using namespace cv;
int main()
{
	//读取数据，分解，转换类型
	FILE* fp;
	fp = fopen("D:\\wendy\\study\\C program\\homework\\week13\\cluster.csv","r");
	if (fp == NULL)
	{
		printf("文件读取失败");
	}//读数据完毕

	char* buffer = (char*)malloc(100 * sizeof(char));
	char* save = buffer;

	int i;
	char seps[] = " ,";
	char* token;
	double number[400];
	int j = 0;
	for (i = 0; i < 200; i++)
	{
		fgets(buffer, 100, fp);
		//printf("%s\t", buffer);//存数据完毕

		token = strtok(buffer, seps);
		while (token != NULL)
		{
			number[j] = atof(token);
			//printf("%f\t", number[j]);
			j++;
			token = strtok(NULL, seps);
		}
		//一组数据分解完毕
	}
	fclose(fp);
	free(buffer);

	for (j = 0; j < 400; j++)
	{
		//printf("%f\n", number[j]);
	}//检验数据读取分解无误

	//得到鱼群数据
	struct fish
	{
		double a = 0;
		double b = 0;
		double distance1 = 0;
		double distance2 = 0;
		int category = 0;
	};
	struct fish fish[200];

	for (i = 0; i < 200; i++)
	{
		fish[i].a = number[i * 2];
		fish[i].b = number[i * 2 + 1];
	}

	//产生初始值
	double x1, x2, y1, y2;//a对应x,b对应y
	srand((unsigned int)time(NULL));
	x1 = rand() % 10 + 10;
	x2 = rand() % 10;
	y1 = rand() % 5 + 5;
	y2 = rand() % 5;
	printf("初始中心坐标为：\t(%f,%f)\t(%f,%f)\n", x1, y1, x2, y2);

	int times = 1;//记录运算次数

	double sum1x = 0;
	double sum1y = 0;
	double sum2x = 0;
	double sum2y = 0;
	int k1 = 0;
	int k2 = 0;

	//开始运算
	//while (1)
	{
		//计算距离
		for (i = 0; i < 200; i++) 
		{
			fish[i].distance1 = sqrt((x1 - fish[i].a) * (x1 - fish[i].a) + (y1 - fish[i].b) * (y1 - fish[i].b));
			fish[i].distance2 = sqrt((x2 - fish[i].a) * (x2 - fish[i].a) + (y2 - fish[i].b) * (y2 - fish[i].b));
			if (fish[i].distance1 < fish[i].distance2)
			{
				fish[i].category = 1;
			}
			if (fish[i].distance1 > fish[i].distance2)
			{
				fish[i].category = 2;
			}//类1为1，类2为2
			//printf("The category of fish %d is %d.\n", i, fish[i].category);
		}

		//分配新的中心
		sum1x = 0;
		sum1y = 0;
		sum2x = 0;
		sum2y = 0;
		k1 = 0;
		k2 = 0;

		for (i = 0; i < 200; i++)
		{
			if (fish[i].category == 1)
			{
				sum1x += fish[i].a;
				sum1y += fish[i].b;
				k1++;
			}
			if (fish[i].category == 2)
			{
				sum2x += fish[i].a;
				sum2y += fish[i].b;
				k2++;
			}
		}
		x1 = (double)sum1x / k1;
		y1 = (double)sum1y / k1;
		x2 = (double)sum2x / k2;
		y2 = (double)sum2y / k2;

		printf("第%d次迭代后中心坐标为：\t(%f,%f)\t(%f,%f)\n", times, x1, y1, x2, y2);



	}


	return 0;
}
